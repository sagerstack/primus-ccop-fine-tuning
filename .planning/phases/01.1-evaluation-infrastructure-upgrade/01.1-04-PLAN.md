2. **Create _score_llm_judge method**:
   ```python
   @staticmethod
   def _score_llm_judge(
       test_case: TestCase,
       response: ModelResponse
   ) -> List[EvaluationMetric]:
       """
       LLM-as-Judge scoring for benchmarks B3, B7-B20.
       Uses benchmark-specific rubric from evaluation-rubrics.md.
       """
       judge_service = LLMJudgeService()

       # Extract benchmark short name (e.g., "B8" from BenchmarkType)
       benchmark_short = test_case.benchmark_type.short_name

       evaluation = judge_service.evaluate_response(
           test_case, response, benchmark_type=benchmark_short
       )

       # Handle judge errors — return empty metrics with judge_error flag
       if evaluation.judge_error:
           return [EvaluationMetric(
               name="accuracy",
               value=0.0,
               weight=1.0,
               description=f"Judge error: {evaluation.error_message}"
           )]

       # Convert JudgeDimensions to EvaluationMetrics
       metrics = []
       for dim in evaluation.dimensions:
           metrics.append(EvaluationMetric(
               name=dim.name,
               value=dim.score / 3.0,  # Normalize 0-3 to 0-1
               weight=dim.weight,
               description=f"LLM judge: {dim.justification[:100]}"
           ))

       # Add overall accuracy metric (for compatibility with existing summary logic)
       metrics.append(EvaluationMetric(
           name="accuracy",
           value=evaluation.overall_score,
           weight=1.0,
           description="LLM judge overall score (weighted dimension average)"
       ))

       return metrics
   ```
