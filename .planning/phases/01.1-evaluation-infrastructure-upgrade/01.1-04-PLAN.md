```python
@staticmethod
def _score_llm_judge(
    test_case: TestCase,
    response: ModelResponse
) -> List[EvaluationMetric]:
    """
    LLM-as-Judge scoring for 15 benchmarks.

    Benchmarks: B3, B7, B8, B9, B10, B11, B12, B13, B14, B15, B16, B17, B18, B19, B20

    Uses Claude as expert judge with benchmark-specific rubrics
    and anchored 0-3 scale from evaluation-rubrics.md.
    """
    judge_service = LLMJudgeService()
    benchmark_short_name = test_case.benchmark_type.short_name  # e.g., "B8"

    evaluation = judge_service.evaluate_response(
        test_case=test_case,
        response=response,
        benchmark_short_name=benchmark_short_name,
    )

    # Handle judge error: return error metrics
    if evaluation.error is not None:
        return [
            EvaluationMetric(
                name="judge_error",
                value=0.0,
                weight=0.0,  # Zero weight so it doesn't affect overall score
                description=f"LLM judge error: {evaluation.error}"
            )
        ]

    # Convert dimension scores to EvaluationMetrics using judge_metric factory
    metrics = []
    for dim in evaluation.dimensions:
        metrics.append(
            judge_metric(
                dimension_name=dim.dimension,
                raw_score=dim.score,
                weight=dim.weight,
            )
        )

    return metrics
```
