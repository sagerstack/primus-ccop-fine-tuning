    **3. Create `_score_llm_judge` method:**

    ```python
    @staticmethod
    def _score_llm_judge(
        test_case: TestCase,
        response: ModelResponse
    ) -> List[EvaluationMetric]:
        """
        LLM-as-Judge scoring for benchmarks B3, B7-B20.

        Uses benchmark-specific rubric prompts from evaluation-rubrics.md
        with 0-3 anchored scale. Returns EvaluationMetric list with
        one metric per dimension.
        """
        # Extract benchmark short name (e.g., "B8")
        benchmark_id = str(test_case.benchmark_type)[:3]
        if hasattr(test_case.benchmark_type, 'short_name'):
            benchmark_id = test_case.benchmark_type.short_name

        judge_service = LLMJudgeService()
        evaluation = judge_service.evaluate_response(test_case, response, benchmark_id)

        # Handle judge errors (skip-and-flag)
        if evaluation.judge_error:
            return [
                EvaluationMetric(
                    name="accuracy",
                    value=0.0,
                    weight=1.0,
                    description=f"judge_error: {evaluation.error_message}"
                )
            ]

        # Convert each dimension to an EvaluationMetric
        metrics = []
        for dim in evaluation.dimensions:
            normalized_score = dim.score / 3.0  # 0-3 -> 0-1
            metrics.append(
                EvaluationMetric(
                    name=dim.name,
                    value=normalized_score,
                    weight=dim.weight,
                    description=f"LLM judge {dim.name} ({dim.score}/3)"
                )
            )

        # If no dimensions returned (shouldn't happen if no error), create fallback
        if not metrics:
            return [
                EvaluationMetric(
                    name="accuracy",
                    value=0.0,
                    weight=1.0,
                    description="judge_error: No dimensions in evaluation"
                )
            ]

        return metrics
    ```
