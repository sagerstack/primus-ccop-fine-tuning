**Part B: Create IExperimentTracker port**

Create `src/application/ports/output/i_experiment_tracker.py` following the same pattern as IModelGateway and IRagPipeline:

```python
"""
Experiment Tracker Port (Interface)

Abstract interface for experiment tracking operations.
Infrastructure layer implements this for specific tracking backends (MLflow, etc.)
"""

from abc import ABC, abstractmethod
from typing import Any, Dict, Optional


class IExperimentTracker(ABC):
    """
    Port (interface) for experiment tracking operations.

    This is an output port - the application depends on this abstraction,
    and the infrastructure provides concrete implementations.
    """

    @abstractmethod
    def start_run(
        self,
        run_name: str,
        params: Dict[str, Any],
    ) -> None:
        """
        Start a new experiment tracking run.

        Args:
            run_name: Unique name for this run (e.g., "primus_llm-only_20260213_143022")
            params: Run parameters to log (model, mode, benchmark_count, etc.)
        """
        pass

    @abstractmethod
    def log_metric(self, key: str, value: float) -> None:
        """
        Log a single metric to the active run.

        Args:
            key: Metric name (e.g., "overall_accuracy", "benchmark_B8")
            value: Metric value (0.0-1.0 for scores)
        """
        pass

    @abstractmethod
    def log_metrics(self, metrics: Dict[str, float]) -> None:
        """
        Log multiple metrics to the active run.

        Args:
            metrics: Dictionary of metric_name -> value
        """
        pass

    @abstractmethod
    def log_artifact(self, file_path: str) -> None:
        """
        Log a file artifact to the active run.

        Args:
            file_path: Path to artifact file (e.g., results JSON)
        """
        pass

    @abstractmethod
    def end_run(self) -> None:
        """End the active experiment tracking run."""
        pass

    @abstractmethod
    def is_available(self) -> bool:
        """
        Check if tracking backend is reachable and configured.

        Returns:
            True if tracking is operational
        """
        pass
```

This port is intentionally simpler than the full MLflow API — it exposes only what the evaluation pipeline needs: start run, log params/metrics/artifacts, end run. No MLflow-specific concepts leak into the application layer.

**What to avoid and WHY:**
- Do NOT expose MLflow-specific types in the port (no mlflow.ActiveRun, no experiment IDs) — the port must be backend-agnostic.
- Do NOT make tracking mandatory — the port is opt-in. If no tracker is configured, the use case should work without it.

Create `src/infrastructure/adapters/tracking/mlflow_tracker.py`:

```python
"""
MLflow Experiment Tracker Adapter

Implementation of IExperimentTracker using MLflow Tracking API.
Integrates with Databricks for experiment storage and comparison UI.
"""

import json
import tempfile
from typing import Any, Dict

import mlflow

from application.ports.output.i_experiment_tracker import IExperimentTracker
from application.ports.output.i_logger import ILogger


class MLflowTracker(IExperimentTracker):
    """
    MLflow implementation of experiment tracking.

    Uses context manager pattern for run lifecycle.
    Requires mlflow_tracking_uri and mlflow_experiment_name in settings.
    """

    def __init__(self, tracking_uri: str, experiment_name: str, logger: ILogger):
        self._tracking_uri = tracking_uri
        self._experiment_name = experiment_name
        self._logger = logger
        self._run_active = False
        self._configured = False

        self._configure()

    def _configure(self) -> None:
        """Configure MLflow tracking URI and experiment."""
        if not self._tracking_uri or not self._experiment_name:
            self._logger.warning(
                "MLflow not configured: missing tracking_uri or experiment_name"
            )
            return

        try:
            mlflow.set_tracking_uri(self._tracking_uri)
            mlflow.set_experiment(self._experiment_name)
            self._configured = True
            self._logger.info(
                f"MLflow configured: uri={self._tracking_uri}, "
                f"experiment={self._experiment_name}"
            )
        except Exception as e:
            self._logger.error(f"Failed to configure MLflow: {e}")
            self._configured = False

    def start_run(self, run_name: str, params: Dict[str, Any]) -> None:
        """Start MLflow run and log parameters."""
        if not self._configured:
            self._logger.warning("MLflow not configured, skipping run start")
            return

        try:
            mlflow.start_run(run_name=run_name)
            self._run_active = True

            # Log all parameters
            for key, value in params.items():
                mlflow.log_param(key, value)

            self._logger.info(f"MLflow run started: {run_name}")
        except Exception as e:
            self._logger.error(f"Failed to start MLflow run: {e}")
            self._run_active = False

    def log_metric(self, key: str, value: float) -> None:
        """Log single metric to active MLflow run."""
        if not self._run_active:
            return
        try:
            mlflow.log_metric(key, value)
        except Exception as e:
            self._logger.error(f"Failed to log metric {key}: {e}")

    def log_metrics(self, metrics: Dict[str, float]) -> None:
        """Log multiple metrics to active MLflow run."""
        if not self._run_active:
            return
        try:
            mlflow.log_metrics(metrics)
        except Exception as e:
            self._logger.error(f"Failed to log metrics: {e}")

    def log_artifact(self, file_path: str) -> None:
        """Log file artifact to active MLflow run."""
        if not self._run_active:
            return
        try:
            mlflow.log_artifact(file_path)
        except Exception as e:
            self._logger.error(f"Failed to log artifact {file_path}: {e}")

    def end_run(self) -> None:
        """End active MLflow run."""
        if not self._run_active:
            return
        try:
            mlflow.end_run()
            self._run_active = False
            self._logger.info("MLflow run ended")
        except Exception as e:
            self._logger.error(f"Failed to end MLflow run: {e}")
            self._run_active = False

    def is_available(self) -> bool:
        """Check if MLflow is configured and reachable."""
        return self._configured
```

Key design decisions:
- Use mlflow.start_run() / mlflow.end_run() (not context manager) because the run lifecycle spans the entire evaluation, and the use case controls start/end.
- All methods are no-ops when not configured or no active run — tracking is optional.
- Errors are logged but don't propagate — tracking failures should never crash evaluation.

**What to avoid and WHY:**
- Do NOT use nested mlflow.start_run() calls — causes parent-child hierarchy confusion (Pitfall 4).
- Do NOT call mlflow.set_experiment() inside start_run — configure once in constructor (Pitfall 4).
- Do NOT make tracking errors fatal — log and continue. Evaluation results matter more than tracking metadata.
